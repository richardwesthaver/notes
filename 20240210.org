* BigBenches
#+name: 1trc-nu
#+begin_src nushell
  let ms = '1trc/measurements-0.parquet'
  dfr open $ms
  | dfr group-by  station
  | dfr agg [
    (dfr col measure | dfr min | dfr as "min")
    (dfr col measure | dfr max | dfr as "max")
    (dfr col measure | dfr sum | dfr as "sum")
    (dfr col measure | dfr count | dfr as "count")
  ]
#+end_src
